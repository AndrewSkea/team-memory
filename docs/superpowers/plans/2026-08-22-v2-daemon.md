# team-memory v2 — Daemon and MCP Server Implementation Plan (Plan 2 of 5)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the single long-lived local process — one binary, one port, one bearer token — that serves MCP over Streamable HTTP, serves the local API the web UI will consume, and cannot phone home.

**Architecture:** One `http.Server` on `127.0.0.1` behind a guard middleware that enforces a Host-header allowlist, an Origin rule, and a constant-time bearer check before any handler runs. MCP is a `POST /mcp` endpoint speaking JSON-RPC over Streamable HTTP at revision `2026-07-28` — no protocol sessions, no `Mcp-Session-Id`, no GET stream. A single-instance lock plus an idle-exit timer means starting the daemon twice is harmless and forgetting to stop it is cheap. Outbound HTTP goes through one shared client whose `DialContext` refuses any host outside an allowlist, making "we never phone home" a property the test suite can assert.

**Tech Stack:** Go 1.26.2, standard library `net/http`. `github.com/AndrewSkea/team-memory/mcp/{paths,store,search,staging,migrate}` from Plan 1. `crypto/subtle` for token comparison. `crypto/rand` for token generation.

**Spec:** `docs/superpowers/specs/2026-08-22-team-memory-v2-design.md` — implements §3 (Runtime architecture), §4 (Security), §7.2 (MCP surface). Read the spec alongside this plan.

## Global Constraints

- Module `github.com/AndrewSkea/team-memory/mcp`. `CGO_ENABLED=0` for every build and test.
- The daemon binds `127.0.0.1` only. Never `0.0.0.0`, never `:80`. The v1 `tryListen(":80")` path and its hosts-file hostname are deleted in this plan and must not return.
- MCP protocol revision is `2026-07-28`. It has **no** protocol sessions: do not emit or require `Mcp-Session-Id`, do not implement `GET /mcp`, do not attempt server-initiated JSON-RPC. Session identity comes from harness hooks (Plan 3), never from MCP.
- Origin rule: **absent → allow. Present and not allowlisted → 403.** Requiring an Origin header breaks clients that do not send one. The spec only requires rejecting a present-and-invalid Origin.
- Bearer tokens are compared with `crypto/subtle.ConstantTimeCompare`, never `==`.
- The token file is mode `0600`. It is never logged, never echoed in an error, and never placed in a URL query string.
- Every outbound HTTP request goes through `net.Client()`. Direct use of `http.DefaultClient`, `http.Get`, or `http.Post` anywhere in the module is a defect.
- No handler blocks on the network while holding a lock.
- Tests use `t.TempDir()` and `httptest`. No test binds a fixed port. No test reaches the network.
- Run tests with `cd mcp && CGO_ENABLED=0 go test ./...`.

```go
// Package netguard provides the only outbound HTTP client in team-memory.
//
// The allowlist lives in DialContext rather than in a request-inspection
// middleware, so it holds regardless of redirects, proxies configured by the
// environment, or a caller who builds their own Request. "We never phone home"
// becomes a property a test can assert rather than a promise in a README.
package netguard

import (
	"context"
	"errors"
	"fmt"
	"net"
	"net/http"
	"strings"
	"sync"
	"time"
)

// ErrBlockedHost is returned when a dial targets a host outside the allowlist.
var ErrBlockedHost = errors.New("netguard: outbound connection to non-allowlisted host blocked")

var defaultHosts = []string{"github.com", "api.github.com", "codeload.github.com"}

var (
	mu     sync.RWMutex
	extra  = map[string]bool{}
	once   sync.Once
	shared *http.Client
)

// allowed returns the effective allowlist.
func allowed() map[string]bool {
	mu.RLock()
	defer mu.RUnlock()
	out := make(map[string]bool, len(defaultHosts)+len(extra))
	for _, h := range defaultHosts {
		out[h] = true
	}
	for h := range extra {
		out[h] = true
	}
	return out
}

// reset clears additions. Test helper.
func reset() {
	mu.Lock()
	defer mu.Unlock()
	extra = map[string]bool{}
}

// Allow adds a host to the allowlist. Call at startup for a self-hosted
// GitHub Enterprise instance.
func Allow(hosts ...string) {
	mu.Lock()
	defer mu.Unlock()
	for _, h := range hosts {
		if h = strings.ToLower(strings.TrimSpace(h)); h != "" {
			extra[h] = true
		}
	}
}

// Client returns the shared allowlist-enforcing HTTP client.
func Client() *http.Client {
	once.Do(func() {
		base := &net.Dialer{Timeout: 15 * time.Second, KeepAlive: 30 * time.Second}
		shared = &http.Client{
			Timeout: 60 * time.Second,
			Transport: &http.Transport{
				Proxy: http.ProxyFromEnvironment,
				DialContext: func(ctx context.Context, network, addr string) (net.Conn, error) {
					host, _, err := net.SplitHostPort(addr)
					if err != nil {
						host = addr
					}
					if !allowed()[strings.ToLower(host)] {
						return nil, fmt.Errorf("%w: %s", ErrBlockedHost, host)
					}
					return base.DialContext(ctx, network, addr)
				},
				ForceAttemptHTTP2:   true,
				MaxIdleConnsPerHost: 4,
				IdleConnTimeout:     90 * time.Second,
			},
		}
	})
	return shared
}
```

```go
import (
	"os"
	"path/filepath"
	"strings"
	"testing"

// TestNoOtherHTTPClientsInModule fails if any package builds its own HTTP
// client or uses the standard library's default one. This is what keeps the
// egress allowlist from being quietly bypassed by a future change.
func TestNoOtherHTTPClientsInModule(t *testing.T) {
	banned := []string{
		"http.DefaultClient",
		"http.Get(",
		"http.Post(",
		"http.PostForm(",
		"http.Head(",
		"&http.Client{",
	}

	root := ".."
	err := filepath.Walk(root, func(path string, info os.FileInfo, err error) error {
		if err != nil {
			return err
		}
		if info.IsDir() {
			if info.Name() == ".git" || info.Name() == "frontend" || info.Name() == "web" {
				return filepath.SkipDir
			}
			return nil
		}
		if !strings.HasSuffix(path, ".go") {
			return nil
		}
		// This package defines the sanctioned client; tests may use httptest.
		if strings.Contains(filepath.ToSlash(path), "/netguard/") {
			return nil
		}
		if strings.HasSuffix(path, "_test.go") {
			return nil
		}

		data, err := os.ReadFile(path)
		if err != nil {
			return err
		}
		for _, b := range banned {
			if strings.Contains(string(data), b) {
				t.Errorf("%s uses %q; route outbound requests through netguard.Client()", path, b)
			}
		}
		return nil
	})
	if err != nil {
		t.Fatalf("walk: %v", err)
	}
}
```

```go
package daemon

import (
	"crypto/rand"
	"crypto/subtle"
	"encoding/base64"
	"fmt"
	"os"
	"strings"

	"github.com/AndrewSkea/team-memory/mcp/paths"
)

const tokenBytes = 32

// LoadOrCreateToken returns the daemon's bearer token, creating it on first
// run. The token is stable across restarts so the web UI does not have to be
// re-authorised every time the daemon starts.
func LoadOrCreateToken(l paths.Layout) (string, error) {
	if raw, err := os.ReadFile(l.TokenFile); err == nil {
		if tok := strings.TrimSpace(string(raw)); tok != "" {
			return tok, nil
		}
	} else if !os.IsNotExist(err) {
		return "", fmt.Errorf("daemon: read token file: %w", err)
	}

	buf := make([]byte, tokenBytes)
	if _, err := rand.Read(buf); err != nil {
		return "", fmt.Errorf("daemon: generate token: %w", err)
	}
	tok := base64.RawURLEncoding.EncodeToString(buf)

	if err := os.WriteFile(l.TokenFile, []byte(tok+"\n"), 0o600); err != nil {
		return "", fmt.Errorf("daemon: write token file: %w", err)
	}
	return tok, nil
}

// TokenMatches compares tokens in constant time. An empty expected token never
// matches, so a failure to load the token can never open the daemon up.
func TokenMatches(want, got string) bool {
	if want == "" {
		return false
	}
	return subtle.ConstantTimeCompare([]byte(want), []byte(got)) == 1
}
```

  - `func (g Guard) Wrap(next http.Handler) http.Handler`

Three checks, in order, before any handler runs:

1. **Host** — must be `127.0.0.1[:port]`, `localhost[:port]`, or `[::1][:port]`. Anything else is 403. This is the primary DNS-rebinding mitigation: an attacker's page resolving `evil.test` to `127.0.0.1` still sends `Host: evil.test`.
2. **Origin** — absent is allowed. Present must be `http://127.0.0.1[:port]`, `http://localhost[:port]`, or `http://[::1][:port]`, else 403.
3. **Bearer** — `Authorization: Bearer <token>`, constant-time compared, else 401.

```go
package daemon

import (
	"net"
	"net/http"
	"net/url"
	"strings"
)

// Guard enforces the daemon's three-part access check. It runs before every
// handler, including the MCP endpoint and the UI's static files.
type Guard struct {
	Token string
}

// localNames are the only host names the daemon answers to.
var localNames = map[string]bool{
	"127.0.0.1": true,
	"localhost": true,
	"::1":       true,
}

// Wrap returns next behind the guard.
func (g Guard) Wrap(next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		// 1. Host. This is the primary DNS-rebinding mitigation: a page served
		// from a hostname that resolves to 127.0.0.1 still sends that hostname
		// in Host, and we refuse it.
		if !isLocalHostname(hostOnly(r.Host)) {
			http.Error(w, "forbidden: unexpected Host header", http.StatusForbidden)
			return
		}

		// 2. Origin. Absent is allowed — CLI harnesses do not send one, and
		// the MCP spec only requires rejecting a present, invalid Origin.
		if origin := r.Header.Get("Origin"); origin != "" && !isLocalOrigin(origin) {
			http.Error(w, "forbidden: unexpected Origin", http.StatusForbidden)
			return
		}

		// 3. Bearer, compared in constant time. The response never repeats
		// either token.
		const prefix = "Bearer "
		auth := r.Header.Get("Authorization")
		if !strings.HasPrefix(auth, prefix) ||
			!TokenMatches(g.Token, strings.TrimSpace(auth[len(prefix):])) {
			w.Header().Set("WWW-Authenticate", `Bearer realm="team-memory"`)
			http.Error(w, "unauthorized", http.StatusUnauthorized)
			return
		}

		next.ServeHTTP(w, r)
	})
}

// hostOnly strips the port and any IPv6 brackets.
func hostOnly(hostport string) string {
	if h, _, err := net.SplitHostPort(hostport); err == nil {
		return strings.Trim(h, "[]")
	}
	return strings.Trim(hostport, "[]")
}

func isLocalHostname(h string) bool {
	return localNames[strings.ToLower(h)]
}

func isLocalOrigin(origin string) bool {
	u, err := url.Parse(origin)
	if err != nil {
		return false
	}
	if u.Scheme != "http" && u.Scheme != "https" {
		return false
	}
	return isLocalHostname(hostOnly(u.Host))
}
```

```go
package daemon

import (
	"encoding/json"
	"fmt"
	"os"
	"path/filepath"
	"time"

	"github.com/AndrewSkea/team-memory/mcp/paths"
)

// Info describes the running daemon. It doubles as the single-instance lock:
// a second `serve` reads it, sees a live process, and connects instead of
// binding a second port.
type Info struct {
	PID     int       `json:"pid"`
	Port    int       `json:"port"`
	Token   string    `json:"token"`
	Started time.Time `json:"started"`
}

func infoPath(l paths.Layout) string { return filepath.Join(l.Home, "daemon.json") }

// WriteInfo records the running daemon. Mode 0600 because it carries the token.
func WriteInfo(l paths.Layout, i Info) error {
	data, err := json.Marshal(i)
	if err != nil {
		return fmt.Errorf("daemon: encode info: %w", err)
	}
	if err := os.WriteFile(infoPath(l), data, 0o600); err != nil {
		return fmt.Errorf("daemon: write info: %w", err)
	}
	return nil
}

// ReadInfo returns the running daemon's details. It returns false when there
// is no lock file or when the recorded process is gone, so a crashed daemon
// never blocks the next start.
func ReadInfo(l paths.Layout) (Info, bool) {
	data, err := os.ReadFile(infoPath(l))
	if err != nil {
		return Info{}, false
	}
	var i Info
	if json.Unmarshal(data, &i) != nil {
		return Info{}, false
	}
	if i.PID <= 0 || !processAlive(i.PID) {
		return Info{}, false
	}
	return i, true
}

// ClearInfo removes the lock file. Safe to call when it does not exist.
func ClearInfo(l paths.Layout) error {
	if err := os.Remove(infoPath(l)); err != nil && !os.IsNotExist(err) {
		return fmt.Errorf("daemon: clear info: %w", err)
	}
	return nil
}
```

```go
		return false
```

```go
package daemon

import (
	"context"
	"errors"
	"fmt"
	"net"
	"net/http"
	"os"
	"sync"
	"time"

	"github.com/AndrewSkea/team-memory/mcp/paths"
)

// DefaultAddr is the daemon's preferred bind address. Loopback only: the
// daemon must never be reachable from another machine.
const DefaultAddr = "127.0.0.1:7373"

// DefaultIdleTimeout stops a forgotten daemon. Thirty minutes is long enough
// that a user switching between sessions never notices a restart.
const DefaultIdleTimeout = 30 * time.Minute

// Options configures Serve.
type Options struct {
	Layout      paths.Layout
	Addr        string
	IdleTimeout time.Duration
	Handler     http.Handler
	OnReady     func(Info)
}

// Serve runs the daemon until ctx is cancelled or the idle timeout elapses.
// It writes the lock file on start and clears it on exit, including on panic
// paths, so a stale file never blocks the next run.
func Serve(ctx context.Context, opt Options) error {
	if opt.Addr == "" {
		opt.Addr = DefaultAddr
	}
	if opt.IdleTimeout == 0 {
		opt.IdleTimeout = DefaultIdleTimeout
	}
	if opt.Handler == nil {
		return errors.New("daemon: Options.Handler is required")
	}
	if err := opt.Layout.EnsureDirs(); err != nil {
		return err
	}

	token, err := LoadOrCreateToken(opt.Layout)
	if err != nil {
		return err
	}

	ln, err := listen(opt.Addr)
	if err != nil {
		return err
	}

	idle := newIdleTimer(opt.IdleTimeout)
	handler := Guard{Token: token}.Wrap(idle.wrap(opt.Handler))
	srv := &http.Server{
		Handler:           handler,
		ReadHeaderTimeout: 10 * time.Second,
		IdleTimeout:       120 * time.Second,
	}

	info := Info{
		PID:     os.Getpid(),
		Port:    ln.Addr().(*net.TCPAddr).Port,
		Token:   token,
		Started: time.Now().UTC(),
	}
	if err := WriteInfo(opt.Layout, info); err != nil {
		ln.Close()
		return err
	}
	defer ClearInfo(opt.Layout)

	if opt.OnReady != nil {
		opt.OnReady(info)
	}

	serveErr := make(chan error, 1)
	go func() {
		err := srv.Serve(ln)
		if errors.Is(err, http.ErrServerClosed) {
			err = nil
		}
		serveErr <- err
	}()

	select {
	case err := <-serveErr:
		return err
	case <-ctx.Done():
	case <-idle.expired():
	}

	shutCtx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
	defer cancel()
	if err := srv.Shutdown(shutCtx); err != nil {
		return fmt.Errorf("daemon: shutdown: %w", err)
	}
	return nil
}

// listen binds the preferred address, falling back to a free port if it is
// taken by something other than us.
func listen(addr string) (net.Listener, error) {
	ln, err := net.Listen("tcp", addr)
	if err == nil {
		return ln, nil
	}
	fallback, ferr := net.Listen("tcp", "127.0.0.1:0")
	if ferr != nil {
		return nil, fmt.Errorf("daemon: listen on %s and on a free port both failed: %w", addr, err)
	}
	return fallback, nil
}

// idleTimer fires when no request has arrived for the configured duration.
type idleTimer struct {
	mu       sync.Mutex
	timer    *time.Timer
	duration time.Duration
	fired    chan struct{}
	once     sync.Once
}

func newIdleTimer(d time.Duration) *idleTimer {
	it := &idleTimer{duration: d, fired: make(chan struct{})}
	it.timer = time.AfterFunc(d, it.fire)
	return it
}

func (it *idleTimer) fire() { it.once.Do(func() { close(it.fired) }) }

func (it *idleTimer) expired() <-chan struct{} { return it.fired }

// wrap resets the countdown on every request.
func (it *idleTimer) wrap(next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		it.mu.Lock()
		it.timer.Reset(it.duration)
		it.mu.Unlock()
		next.ServeHTTP(w, r)
	})
}
```

```go
// Package rpc implements the JSON-RPC 2.0 envelope. It is transport-agnostic:
// the same types serve the HTTP endpoint and the stdio fallback.
package rpc

import (
	"bytes"
	"encoding/json"
)

// Standard JSON-RPC 2.0 error codes.
const (
	CodeParse          = -32700
	CodeInvalidRequest = -32600
	CodeMethodNotFound = -32601
	CodeInvalidParams  = -32602
	CodeInternal       = -32603
)

// Request is an incoming JSON-RPC call.
type Request struct {
	JSONRPC string          `json:"jsonrpc"`
	ID      json.RawMessage `json:"id,omitempty"`
	Method  string          `json:"method"`
	Params  json.RawMessage `json:"params,omitempty"`
}

// IsNotification reports whether the caller expects no response.
func (r Request) IsNotification() bool {
	return len(r.ID) == 0 || bytes.Equal(bytes.TrimSpace(r.ID), []byte("null"))
}

// Error is a JSON-RPC error object.
type Error struct {
	Code    int    `json:"code"`
	Message string `json:"message"`
	Data    any    `json:"data,omitempty"`
}

// Response is an outgoing JSON-RPC reply. Exactly one of Result and Error is
// set, enforced by the constructors below.
type Response struct {
	JSONRPC string          `json:"jsonrpc"`
	ID      json.RawMessage `json:"id,omitempty"`
	Result  any             `json:"result,omitempty"`
	Error   *Error          `json:"error,omitempty"`
}

// Result builds a success response.
func Result(id json.RawMessage, v any) Response {
	return Response{JSONRPC: "2.0", ID: id, Result: v}
}

// Fail builds an error response.
func Fail(id json.RawMessage, code int, msg string) Response {
	return Response{JSONRPC: "2.0", ID: id, Error: &Error{Code: code, Message: msg}}
}
```

```go
// Package mcpserver implements the Model Context Protocol over Streamable
// HTTP at revision 2026-07-28.
//
// That revision removed protocol sessions, the Mcp-Session-Id header, the GET
// event stream, and server-initiated JSON-RPC. This server therefore answers
// POST only, is stateless between calls, and never emits a session header.
// Session identity comes from harness hooks, not from MCP.
package mcpserver

import (
	"context"
	"encoding/json"
	"io"
	"net/http"

	"github.com/AndrewSkea/team-memory/mcp/rpc"
	"github.com/AndrewSkea/team-memory/mcp/search"
	"github.com/AndrewSkea/team-memory/mcp/staging"
	"github.com/AndrewSkea/team-memory/mcp/store"
)

// ProtocolVersion is the MCP revision this server speaks.
const ProtocolVersion = "2026-07-28"

// maxBody caps a single JSON-RPC request. A local client has no reason to
// exceed this, and the cap keeps a malformed request from exhausting memory.
const maxBody = 4 << 20 // 4 MiB

// Deps are the storage components the tools operate on.
type Deps struct {
	Store   *store.Store
	Index   *search.Index
	Staging *staging.Area
}

// Server dispatches MCP methods.
type Server struct {
	deps Deps
}

// New builds a Server.
func New(d Deps) *Server { return &Server{deps: d} }

// Dispatch routes one JSON-RPC request. The second return is false for
// notifications, which get no response.
func (s *Server) Dispatch(ctx context.Context, req rpc.Request) (rpc.Response, bool) {
	if req.IsNotification() {
		return rpc.Response{}, false
	}
	switch req.Method {
	case "initialize":
		return s.initialize(req), true
	case "tools/list":
		return rpc.Result(req.ID, map[string]any{"tools": toolDescriptors()}), true
	case "tools/call":
		return s.callTool(ctx, req), true
	case "ping":
		return rpc.Result(req.ID, map[string]any{}), true
	default:
		return rpc.Fail(req.ID, rpc.CodeMethodNotFound, "unknown method: "+req.Method), true
	}
}

func (s *Server) initialize(req rpc.Request) rpc.Response {
	return rpc.Result(req.ID, map[string]any{
		"protocolVersion": ProtocolVersion,
		"capabilities": map[string]any{
			"tools": map[string]any{"listChanged": false},
		},
		"serverInfo": map[string]any{
			"name":    "team-memory",
			"version": "2.0.0",
		},
	})
}

// Handler serves POST /mcp. GET is 405: the streaming endpoint no longer
// exists in this protocol revision.
func (s *Server) Handler() http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		if r.Method != http.MethodPost {
			w.Header().Set("Allow", http.MethodPost)
			http.Error(w, "method not allowed", http.StatusMethodNotAllowed)
			return
		}

		body, err := io.ReadAll(io.LimitReader(r.Body, maxBody))
		if err != nil {
			writeJSON(w, http.StatusOK,
				rpc.Fail(nil, rpc.CodeParse, "could not read request body"))
			return
		}

		var req rpc.Request
		if err := json.Unmarshal(body, &req); err != nil {
			writeJSON(w, http.StatusOK, rpc.Fail(nil, rpc.CodeParse, "invalid JSON"))
			return
		}

		resp, ok := s.Dispatch(r.Context(), req)
		if !ok {
			w.WriteHeader(http.StatusAccepted)
			return
		}
		writeJSON(w, http.StatusOK, resp)
	})
}

func writeJSON(w http.ResponseWriter, status int, v any) {
	w.Header().Set("Content-Type", "application/json")
	w.WriteHeader(status)
	_ = json.NewEncoder(w).Encode(v)
}
```

```go
// callParams is the tools/call parameter shape.
type callParams struct {
	Name      string          `json:"name"`
	Arguments json.RawMessage `json:"arguments"`
}

func (s *Server) callTool(ctx context.Context, req rpc.Request) rpc.Response {
	var p callParams
	if err := json.Unmarshal(req.Params, &p); err != nil {
		return rpc.Fail(req.ID, rpc.CodeInvalidParams, "params must be an object with name and arguments")
	}
	switch p.Name {
	case "search_memory":
		return s.toolSearch(req, p.Arguments)
	case "get_memory":
		return s.toolGet(req, p.Arguments)
	case "remember":
		return s.toolRemember(req, p.Arguments)
	default:
		return rpc.Fail(req.ID, rpc.CodeInvalidParams, "unknown tool: "+p.Name)
	}
}

// text builds a successful tool result.
func text(id json.RawMessage, body string) rpc.Response {
	return rpc.Result(id, map[string]any{
		"content": []map[string]any{{"type": "text", "text": body}},
	})
}

// toolError builds a tool-level failure. This is distinct from a JSON-RPC
// error: the call succeeded, the tool reported a problem, and the model can
// read it and try something else.
func toolError(id json.RawMessage, body string) rpc.Response {
	return rpc.Result(id, map[string]any{
		"content": []map[string]any{{"type": "text", "text": body}},
		"isError": true,
	})
}

func (s *Server) toolSearch(req rpc.Request, args json.RawMessage) rpc.Response {
	var a struct {
		Query string `json:"query"`
		Limit int    `json:"limit"`
	}
	if err := json.Unmarshal(args, &a); err != nil {
		return rpc.Fail(req.ID, rpc.CodeInvalidParams, "arguments must include a query string")
	}
	if strings.TrimSpace(a.Query) == "" {
		return rpc.Fail(req.ID, rpc.CodeInvalidParams, "query must not be empty")
	}
	if a.Limit <= 0 {
		a.Limit = 10
	}

	hits, err := s.deps.Index.Search(a.Query, a.Limit)
	if err != nil {
		return toolError(req.ID, "search failed: "+err.Error())
	}
	if len(hits) == 0 {
		return text(req.ID, "No memories match that query.")
	}

	var b strings.Builder
	fmt.Fprintf(&b, "%d matching memories:\n\n", len(hits))
	for _, h := range hits {
		fmt.Fprintf(&b, "- %s (%s) %s\n  %s\n", h.ID, h.Scope, h.Slug, h.Snippet)
	}
	b.WriteString("\nUse get_memory with an id to read the full text.")
	return text(req.ID, b.String())
}

func (s *Server) toolGet(req rpc.Request, args json.RawMessage) rpc.Response {
	var a struct {
		ID string `json:"id"`
	}
	if err := json.Unmarshal(args, &a); err != nil || strings.TrimSpace(a.ID) == "" {
		return rpc.Fail(req.ID, rpc.CodeInvalidParams, "arguments must include an id string")
	}

	all, err := s.deps.Store.All()
	if err != nil {
		return toolError(req.ID, "could not read memories: "+err.Error())
	}
	for _, m := range all {
		if m.ID == a.ID {
			return text(req.ID, fmt.Sprintf("%s (%s, %s)\n\n%s",
				m.Slug, m.Scope, m.Created.Format("2006-01-02"), m.Body))
		}
	}
	return toolError(req.ID, "No memory has id "+a.ID+". Use search_memory to find valid ids.")
}

func (s *Server) toolRemember(req rpc.Request, args json.RawMessage) rpc.Response {
	var a struct {
		Text  string   `json:"text"`
		Title string   `json:"title"`
		Scope string   `json:"scope"`
		Tags  []string `json:"tags"`
	}
	if err := json.Unmarshal(args, &a); err != nil {
		return rpc.Fail(req.ID, rpc.CodeInvalidParams, "arguments must include a text string")
	}
	if strings.TrimSpace(a.Text) == "" {
		return rpc.Fail(req.ID, rpc.CodeInvalidParams, "text must not be empty")
	}

	title := a.Title
	if strings.TrimSpace(title) == "" {
		title = firstSentence(a.Text)
	}
	scope := memory.ScopeTeam
	if strings.EqualFold(a.Scope, string(memory.ScopePersonal)) {
		scope = memory.ScopePersonal
	}

	m := memory.Memory{
		Created: time.Now().UTC().Truncate(time.Second),
		Author:  currentUser(),
		Scope:   scope,
		Tags:    a.Tags,
		Slug:    title,
		Body:    strings.TrimSpace(a.Text),
	}
	m.ID = memory.ComputeID(m)

	if err := s.deps.Staging.Add(m); err != nil {
		return toolError(req.ID, "could not queue the memory: "+err.Error())
	}
	return text(req.ID,
		"Queued for review as "+m.ID+". It is not in the team repository until it is approved.")
}

// firstSentence takes a title from the opening sentence of the body.
func firstSentence(s string) string {
	s = strings.TrimSpace(s)
	if i := strings.IndexAny(s, ".!?\n"); i > 0 {
		s = s[:i]
	}
	if len(s) > 80 {
		s = s[:80]
	}
	return s
}

// currentUser names the author of a staged memory. It is a local identity
// only; nothing is sent anywhere.
func currentUser() string {
	for _, key := range []string{"TEAM_MEMORY_USER", "USER", "USERNAME"} {
		if v := strings.TrimSpace(os.Getenv(key)); v != "" {
			return v
		}
	}
	return "unknown"
}
```

```go
// Run executes one command and returns the process exit code.
func Run(args []string, stdout, stderr io.Writer) int {
	if len(args) == 0 {
		fmt.Fprint(stdout, usage)

	}
	switch args[0] {
	case "help", "-h", "--help":
		fmt.Fprint(stdout, usage)
		return 0
	case "version", "--version":
		fmt.Fprintf(stdout, "team-memory %s\n", Version)
		return 0
	case "serve":
		return cmdServe(stdout, stderr)
	case "status":
		return cmdStatus(stdout, stderr)
	case "stop":
		return cmdStop(stdout, stderr)
	case "migrate":
		return cmdMigrate(stdout, stderr)
	default:
		fmt.Fprintf(stderr, "team-memory: unknown command %q\n\n%s", args[0], usage)
		return 2
	}
}

// openAll resolves the layout and opens every storage component.
func openAll() (paths.Layout, *store.Store, *search.Index, *staging.Area, error) {
	l, err := paths.Resolve()
	if err != nil {
		return paths.Layout{}, nil, nil, nil, err
	}
	st, err := store.Open(l)
	if err != nil {
		return l, nil, nil, nil, err
	}
	ix, err := search.Open(l.Index)
	if err != nil {
		return l, nil, nil, nil, err
	}
	sa, err := staging.New(l)
	if err != nil {
		return l, nil, nil, nil, err
	}
	return l, st, ix, sa, nil
}

func cmdServe(stdout, stderr io.Writer) int {
	l, st, ix, sa, err := openAll()
	if err != nil {
		fmt.Fprintf(stderr, "team-memory: %v\n", err)
		return 1
	}
	defer ix.Close()

	// A daemon is already up: say where, and exit successfully. Starting
	// twice must be harmless, because every hook invocation tries.
	if info, running := daemon.ReadInfo(l); running {
		fmt.Fprintf(stdout, "team-memory already running on http://127.0.0.1:%d (pid %d)\n",
			info.Port, info.PID)
		return 0
	}

	all, err := st.All()
	if err != nil {
		fmt.Fprintf(stderr, "team-memory: %v\n", err)
		return 1
	}
	if err := ix.Rebuild(all); err != nil {
		fmt.Fprintf(stderr, "team-memory: build search index: %v\n", err)
		return 1
	}

	mux := http.NewServeMux()
	mux.Handle("/mcp", mcpserver.New(mcpserver.Deps{Store: st, Index: ix, Staging: sa}).Handler())
	mux.HandleFunc("/healthz", func(w http.ResponseWriter, r *http.Request) {
		fmt.Fprintln(w, "ok")
	})

	ctx, stop := signal.NotifyContext(context.Background(), os.Interrupt, syscall.SIGTERM)
	defer stop()

	err = daemon.Serve(ctx, daemon.Options{
		Layout:  l,
		Handler: mux,
		OnReady: func(i daemon.Info) {
			fmt.Fprintf(stdout, "team-memory listening on http://127.0.0.1:%d\n", i.Port)
		},
	})
	if err != nil {
		fmt.Fprintf(stderr, "team-memory: %v\n", err)
		return 1
	}
	return 0
}

func cmdStatus(stdout, stderr io.Writer) int {
	l, err := paths.Resolve()
	if err != nil {
		fmt.Fprintf(stderr, "team-memory: %v\n", err)
		return 1
	}
	info, running := daemon.ReadInfo(l)
	if !running {
		fmt.Fprintln(stdout, "team-memory is not running. Start it with: team-memory serve")
		return 1
	}
	fmt.Fprintf(stdout, "team-memory is running on http://127.0.0.1:%d (pid %d, since %s)\n",
		info.Port, info.PID, info.Started.Local().Format("15:04:05"))
	return 0
}

func cmdStop(stdout, stderr io.Writer) int {
	l, err := paths.Resolve()
	if err != nil {
		fmt.Fprintf(stderr, "team-memory: %v\n", err)
		return 1
	}
	info, running := daemon.ReadInfo(l)
	if !running {
		fmt.Fprintln(stdout, "team-memory is not running.")
		return 0
	}
	p, err := os.FindProcess(info.PID)
	if err != nil {
		fmt.Fprintf(stderr, "team-memory: find process %d: %v\n", info.PID, err)
		return 1
	}
	if err := p.Signal(os.Interrupt); err != nil {
		fmt.Fprintf(stderr, "team-memory: stop process %d: %v\n", info.PID, err)
		return 1
	}
	fmt.Fprintf(stdout, "Stopped team-memory (pid %d).\n", info.PID)
	return 0
}

func cmdMigrate(stdout, stderr io.Writer) int {
	_, st, ix, sa, err := openAll()
	if err != nil {
		fmt.Fprintf(stderr, "team-memory: %v\n", err)
		return 1
	}
	defer ix.Close()

	res, err := migrate.Run(st, sa)
	if err != nil {
		fmt.Fprintf(stderr, "team-memory: migrate: %v\n", err)
		return 1
	}
	fmt.Fprintf(stdout, "Migrated %d memories. %d queued for review. %d preserved unparsed.\n",
		res.Converted, res.Pending, res.Unparsed)
	if res.Unparsed > 0 {
		fmt.Fprintln(stdout,
			"Unparsed entries are kept verbatim under memories/team/unparsed/ — nothing was discarded.")
	}
	return 0
}
```
