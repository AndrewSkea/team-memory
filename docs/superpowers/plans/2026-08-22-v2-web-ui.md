# team-memory v2 — Web UI Implementation Plan (Plan 4 of 5)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Give the user a local web page where they can see exactly what team-memory knows about them, choose how much it is allowed to save, approve or reject each candidate memory, and recover anything they threw away by mistake.

**Architecture:** A SvelteKit app built to static files with `adapter-static`, embedded into the Go binary with `go:embed`, and served by the same daemon that serves `/api/` and `/mcp`. One process, one port, no build step at run time. The page is opened by `team-memory ui`, which starts the daemon if needed and opens a URL carrying the bearer token once; the app stores the token in memory for the session and never writes it to `localStorage`.

**Tech Stack:** Svelte 5 (runes), SvelteKit with `@sveltejs/adapter-static`, TypeScript, Vite, Vitest with `@testing-library/svelte` and `jsdom`. No component library, no CSS framework, no CDN.

**Spec:** `docs/superpowers/specs/2026-08-22-team-memory-v2-design.md` — implements §8.1 and §8.2. Read the spec alongside this plan.

## Global Constraints

- The app must work with no network access at all. No CDN, no web font fetch, no analytics, no error reporting. Everything ships inside the binary. A page that phones home contradicts the product.
- Static output only: `adapter-static` with `fallback: 'index.html'` and `ssr = false`. There is no Node server at run time.
- The bearer token lives in a module-level variable. Never `localStorage`, never `sessionStorage`, never a cookie, never a query string that survives in the address bar.
- Every list must render an empty state. An empty review queue is the normal, healthy state and must look like success rather than an error.
- Copy is plain and honest. "Delete permanently" is only used where deletion really is permanent; a memory that was pushed says so instead.
- Node 22 or later. `npm ci`, never `npm install`, in CI.
- Frontend tests run with `cd web && npm test`. Go tests still run with `cd mcp && CGO_ENABLED=0 go test ./...`.

```ts
/** Formatting shared by more than one view. */

const MS_PER_DAY = 86_400_000;

/**
 * relativeDate renders an ISO timestamp the way a person would say it.
 * Anything older than a fortnight gets a real date, because "37 days ago"
 * is harder to place than "1 May 2026".
 */
export function relativeDate(iso: string, now: Date = new Date()): string {
  const then = new Date(iso);
  if (Number.isNaN(then.getTime())) return 'unknown';

  const startOf = (d: Date) => Date.UTC(d.getUTCFullYear(), d.getUTCMonth(), d.getUTCDate());
  const days = Math.round((startOf(now) - startOf(then)) / MS_PER_DAY);

  if (days <= 0) return 'today';
  if (days === 1) return 'yesterday';
  if (days < 14) return `${days} days ago`;

  return then.toLocaleDateString('en-GB', {
    day: 'numeric',
    month: 'long',
    year: 'numeric',
    timeZone: 'UTC'
  });
}

/** truncate cuts at a word boundary so the result never ends mid-word. */
export function truncate(text: string, max: number): string {
  if (text.length <= max) return text;
  const cut = text.slice(0, max);
  const lastSpace = cut.lastIndexOf(' ');
  return (lastSpace > 0 ? cut.slice(0, lastSpace) : cut) + '…';
}

/** titleFor always yields something displayable. */
export function titleFor(m: { title?: string; body?: string }): string {
  const title = (m.title ?? '').trim();
  if (title) return title;

  const body = (m.body ?? '').trim();
  if (!body) return 'Untitled memory';

  const end = body.search(/[.!?\n]/);
  const first = (end > 0 ? body.slice(0, end) : body).trim();
  return truncate(first || 'Untitled memory', 80);
}
```

```ts
/**
 * Every call to the local Go API lives here.
 *
 * The bearer token is held in a module variable for the life of the page. It
 * is deliberately not written to localStorage or a cookie: a token that
 * survives the tab is a token that can be read by anything else the browser
 * later runs on this origin.
 */

export type Scope = 'team' | 'personal';

export interface Memory {
  id: string;
  title: string;
  body: string;
  scope: Scope;
  author: string;
  tags: string[];
  created: string;
  source_session?: string;
  confidence?: number;
}

export interface BinItem {
  id: string;
  title: string;
  body: string;
  reason: string;
  rejected: string;
}

export interface Hit {
  id: string;
  slug: string;
  scope: string;
  snippet: string;
}

export interface Settings {
  level: 'off' | 'review' | 'auto';
  repo: string;
  author: string;
  agent: string;
  pushes: boolean;
}

export interface Edits {
  title?: string;
  body?: string;
  scope?: Scope;
  tags?: string[];
}

export class ApiError extends Error {
  status: number;
  constructor(message: string, status: number) {
    super(message);
    this.name = 'ApiError';
    this.status = status;
  }
}

let token = '';

export function setToken(t: string): void {
  token = t;
}

export function hasToken(): boolean {
  return token !== '';
}

async function call<T>(path: string, init: RequestInit = {}): Promise<T> {
  let res: Response;
  try {
    res = await fetch(path, {
      ...init,
      headers: {
        'Content-Type': 'application/json',
        Authorization: `Bearer ${token}`,
        ...(init.headers ?? {})
      }
    });
  } catch {
    // A fetch rejection here almost always means the daemon stopped, so say
    // that rather than surfacing "Failed to fetch".
    throw new ApiError(
      'Cannot reach team-memory on this computer. It may have shut down — run `team-memory serve` and reload.',
      0
    );
  }

  if (!res.ok) {
    let message = res.statusText || `request failed with ${res.status}`;
    try {
      const body = await res.json();
      if (body && typeof body.error === 'string') message = body.error;
    } catch {
      // Not JSON. Keep the status text.
    }
    throw new ApiError(message, res.status);
  }

  if (res.status === 204) return undefined as T;
  return (await res.json()) as T;
}

function post<T>(path: string, body: unknown): Promise<T> {
  return call<T>(path, { method: 'POST', body: JSON.stringify(body) });
}

export async function getPending(): Promise<Memory[]> {
  const r = await call<{ pending: Memory[] | null }>('/api/pending');
  return r.pending ?? [];
}

export async function getMemories(): Promise<Memory[]> {
  const r = await call<{ memories: Memory[] | null }>('/api/memories');
  return r.memories ?? [];
}

export async function getBin(): Promise<BinItem[]> {
  const r = await call<{ bin: BinItem[] | null }>('/api/bin');
  return r.bin ?? [];
}

export async function searchMemories(q: string): Promise<Hit[]> {
  const query = q.trim();
  if (!query) return [];
  const r = await call<{ hits: Hit[] | null }>(`/api/search?q=${encodeURIComponent(query)}`);
  return r.hits ?? [];
}

export function approve(id: string, edits: Edits = {}): Promise<{ id: string; status: string }> {
  return post('/api/approve', { id, ...edits });
}

export function reject(id: string, reason?: string) {
  return post<{ id: string; status: string }>('/api/reject', reason ? { id, reason } : { id });
}

export function restore(id: string) {
  return post<{ id: string; status: string }>('/api/restore', { id });
}

export function purge(id: string) {
  return post<{ id: string; status: string; note: string }>('/api/purge', { id });
}

export function forget(id: string) {
  return post<{ id: string; status: string; note: string }>('/api/forget', { id });
}

export function getSettings(): Promise<Settings> {
  return call<Settings>('/api/settings');
}

export function setLevel(level: Settings['level']) {
  return post<Settings>('/api/settings/level', { level });
}

export function setRepo(url: string, pat: string) {
  return post<Settings>('/api/settings/repo', { url, token: pat });
}
```

Note the query-encoding test expects `%20` for spaces, which
`encodeURIComponent` produces. If a `+` is wanted instead, the Go side must be
changed too; leave both as they are.

```ts
import type { Settings } from './api';

/**
 * appState is the small amount of state more than one route needs: how many
 * memories are waiting, what the memory level is, and the last error worth
 * showing. Everything else belongs to its own route.
 */
class AppState {
  pendingCount = $state(0);
  level = $state<Settings['level']>('off');
  error = $state('');

  fail(message: string) {
    this.error = message;
  }

  clearError() {
    this.error = '';
  }

  reset() {
    this.pendingCount = 0;
    this.level = 'off';
    this.error = '';
  }
}

export const appState = new AppState();
```

```ts
/**
 * takeTokenFromUrl reads the bearer token the CLI put in the URL fragment.
 *
 * Only the fragment is accepted. A fragment is never sent to a server, never
 * appears in an access log, and is easy to clear from the address bar; a query
 * string is none of those things.
 */
export function takeTokenFromUrl(loc: { hash: string; search: string }): string {
  const hash = loc.hash.startsWith('#') ? loc.hash.slice(1) : loc.hash;
  if (!hash) return '';
  const value = new URLSearchParams(hash).get('token');
  return value ?? '';
}
```

```svelte
<script lang="ts">
  import { approve, getPending, reject, type Edits, type Memory } from '$lib/api';
  import Empty from '$lib/components/Empty.svelte';
  import MemoryCard from '$lib/components/MemoryCard.svelte';
  import { appState } from '$lib/state.svelte';

  let items = $state<Memory[]>([]);
  let loading = $state(true);
  let working = $state(false);

  async function load() {
    loading = true;
    try {
      items = await getPending();
      appState.pendingCount = items.length;
    } catch (err) {
      appState.fail(err instanceof Error ? err.message : String(err));
    } finally {
      loading = false;
    }
  }

  $effect(() => {
    void load();
  });

  /**
   * act removes the card immediately and puts it back if the call fails.
   * The optimistic removal is what makes a queue of ten feel quick; the
   * restore is what stops a failure quietly losing someone's memory.
   */
  async function act(id: string, run: () => Promise<unknown>) {
    const index = items.findIndex((m) => m.id === id);
    if (index < 0) return;
    const removed = items[index];

    items = items.filter((m) => m.id !== id);
    appState.pendingCount = items.length;

    try {
      await run();
    } catch (err) {
      items = [...items.slice(0, index), removed, ...items.slice(index)];
      appState.pendingCount = items.length;
      appState.fail(err instanceof Error ? err.message : String(err));
    }
  }

  function onapprove(id: string, edits: Edits) {
    void act(id, () => approve(id, edits));
  }

  function onreject(id: string) {
    void act(id, () => reject(id));
  }

  async function approveAll() {
    working = true;
    const snapshot = items;
    try {
      // Sequential rather than parallel: each approval is a git commit, and
      // committing concurrently in one repository fails.
      for (const m of snapshot) {
        await approve(m.id, {});
      }
      items = [];
      appState.pendingCount = 0;
    } catch (err) {
      appState.fail(err instanceof Error ? err.message : String(err));
      await load();
    } finally {
      working = false;
    }
  }
</script>
```

```go
package review

import (
	"encoding/json"
	"fmt"
	"net/http"
	"os"
	"path/filepath"
	"strings"

	"github.com/AndrewSkea/team-memory/mcp/capture"
	"github.com/AndrewSkea/team-memory/mcp/extract"
)

// settingsView is what the UI is allowed to see. There is deliberately no
// token field: the credential goes in and never comes back out.
type settingsView struct {
	Level  string `json:"level"`
	Repo   string `json:"repo"`
	Author string `json:"author"`
	Agent  string `json:"agent"`
	Pushes bool   `json:"pushes"`
}

func (a *API) settings(w http.ResponseWriter, r *http.Request) {
	writeJSON(w, http.StatusOK, a.currentSettings())
}

func (a *API) currentSettings() settingsView {
	repo, _ := a.store.Remote()

	agent := ""
	if found := extract.Detect(); len(found) > 0 {
		agent = found[0].Name()
	}

	return settingsView{
		Level:  a.readLevel(),
		Repo:   repo,
		Author: localAuthor(),
		Agent:  agent,
		Pushes: repo != "",
	}
}

// readLevel fails closed: anything unreadable or unrecognised is "off".
func (a *API) readLevel() string {
	data, err := os.ReadFile(filepath.Join(a.layout.Home, "level"))
	if err != nil {
		return capture.LevelOff
	}
	switch level := strings.TrimSpace(string(data)); level {
	case capture.LevelReview, capture.LevelAuto:
		return level
	default:
		return capture.LevelOff
	}
}

func (a *API) setLevel(w http.ResponseWriter, r *http.Request) {
	var req struct {
		Level string `json:"level"`
	}
	if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
		writeError(w, http.StatusBadRequest, "body must be an object with a level")
		return
	}
	switch req.Level {
	case capture.LevelOff, capture.LevelReview, capture.LevelAuto:
	default:
		writeError(w, http.StatusBadRequest,
			fmt.Sprintf("level must be one of %q, %q or %q",
				capture.LevelOff, capture.LevelReview, capture.LevelAuto))
		return
	}

	path := filepath.Join(a.layout.Home, "level")
	if err := os.WriteFile(path, []byte(req.Level+"\n"), 0o600); err != nil {
		writeError(w, http.StatusInternalServerError, err.Error())
		return
	}
	writeJSON(w, http.StatusOK, a.currentSettings())
}

func (a *API) setRepo(w http.ResponseWriter, r *http.Request) {
	var req struct {
		URL   string `json:"url"`
		Token string `json:"token"`
	}
	if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
		writeError(w, http.StatusBadRequest, "body must be an object with a url")
		return
	}

	url := strings.TrimSpace(req.URL)
	if url == "" {
		writeError(w, http.StatusBadRequest, "a repository url is required")
		return
	}
	if err := a.store.SetRemote(url); err != nil {
		writeError(w, http.StatusInternalServerError, err.Error())
		return
	}

	// A blank token means "keep what is already there", which is what the
	// placeholder in the form promises.
	if token := strings.TrimSpace(req.Token); token != "" {
		if err := os.WriteFile(a.layout.GitTokenFile, []byte(token+"\n"), 0o600); err != nil {
			writeError(w, http.StatusInternalServerError, err.Error())
			return
		}
	}

	writeJSON(w, http.StatusOK, a.currentSettings())
}

// localAuthor names the user for attribution. Local identity only.
func localAuthor() string {
	for _, key := range []string{"TEAM_MEMORY_USER", "USER", "USERNAME"} {
		if v := strings.TrimSpace(os.Getenv(key)); v != "" {
			return v
		}
	}
	return "unknown"
}
```

```go
// Package webui serves the built Svelte app out of the binary.
//
// The bundle is copied into build/ by scripts/build.sh before `go build`
// runs. Embedding it means there is one file to install and no run-time build
// step, which is the whole point of shipping a single binary.
package webui

import (
	"embed"
	"io/fs"
	"net/http"
	"strings"
)

//go:embed all:build
var bundle embed.FS

// content is build/ rooted at its top level, or nil if nothing was embedded.
var content fs.FS = func() fs.FS {
	sub, err := fs.Sub(bundle, "build")
	if err != nil {
		return nil
	}
	if _, err := fs.Stat(sub, "index.html"); err != nil {
		return nil
	}
	return sub
}()

// Present reports whether this build carries a real UI bundle.
func Present() bool { return content != nil }

const noBundle = `team-memory was built without its web interface.

Build it with:

    scripts/build.sh

which runs the frontend build and then compiles the binary with the result
embedded.
`

// Handler serves the app, falling back to index.html for client-side routes.
func Handler() http.Handler {
	if !Present() {
		return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
			w.Header().Set("Content-Type", "text/plain; charset=utf-8")
			w.WriteHeader(http.StatusOK)
			_, _ = w.Write([]byte(noBundle))
		})
	}

	files := http.FileServer(http.FS(content))

	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		name := strings.TrimPrefix(r.URL.Path, "/")

		// Hashed assets are immutable; the index is not, or a stale copy
		// would keep loading a bundle that no longer exists.
		if strings.HasPrefix(name, "_app/immutable/") {
			w.Header().Set("Cache-Control", "public, max-age=31536000, immutable")
			files.ServeHTTP(w, r)
			return
		}

		if name != "" {
			if _, err := fs.Stat(content, name); err == nil {
				w.Header().Set("Cache-Control", "no-store")
				files.ServeHTTP(w, r)
				return
			}
		}

		serveIndex(w)
	})
}

func serveIndex(w http.ResponseWriter) {
	data, err := fs.ReadFile(content, "index.html")
	if err != nil {
		http.Error(w, "index.html is missing from the embedded bundle", http.StatusInternalServerError)
		return
	}
	w.Header().Set("Content-Type", "text/html; charset=utf-8")
	w.Header().Set("Cache-Control", "no-store")
	w.WriteHeader(http.StatusOK)
	_, _ = w.Write(data)
}
```

```go
// cmdUI opens the web page, starting the daemon if it is not already up.
//
// The token goes in the URL fragment, which is never sent to a server and
// never appears in a log, and the page clears it from the address bar as soon
// as it has read it.
func cmdUI(stdout, stderr io.Writer) int {
	l, err := paths.Resolve()
	if err != nil {
		fmt.Fprintf(stderr, "team-memory: %v\n", err)
		return 1
	}

	if _, running := daemon.ReadInfo(l); !running {
		if err := spawnDaemon(); err != nil {
			fmt.Fprintf(stderr, "team-memory: could not start: %v\n", err)
			return 1
		}
		waitForDaemon(l, 5*time.Second)
	}

	info, running := daemon.ReadInfo(l)
	if !running {
		fmt.Fprintln(stderr, "team-memory: the daemon did not start")
		return 1
	}

	url := fmt.Sprintf("http://127.0.0.1:%d/#token=%s",
		info.Port, neturl.QueryEscape(info.Token))

	if err := openBrowser(url); err != nil {
		fmt.Fprintf(stdout, "Open this in your browser:\n\n  %s\n", url)
		return 0
	}
	fmt.Fprintf(stdout, "Opened team-memory on port %d.\n", info.Port)
	return 0
}
```

```go
package cli

import (
	"fmt"
	"os/exec"
	"runtime"
)

// openBrowser opens a URL with whatever the platform uses for the job.
func openBrowser(url string) error {
	var cmd *exec.Cmd
	switch runtime.GOOS {
	case "darwin":
		cmd = exec.Command("open", url)
	case "windows":
		// rundll32 avoids cmd.exe's handling of & in a URL.
		cmd = exec.Command("rundll32", "url.dll,FileProtocolHandler", url)
	default:
		cmd = exec.Command("xdg-open", url)
	}
	if err := cmd.Start(); err != nil {
		return fmt.Errorf("open browser: %w", err)
	}
	return cmd.Process.Release()
}
```
